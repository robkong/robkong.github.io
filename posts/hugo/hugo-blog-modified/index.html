修改网站图标
#

推荐网站 favicon.io 上传图片就能获得图标文件，将下载的文件放在根目录的 static 文件中。

如果图标不显示，清除浏览器缓存后，重启服务。

代码块隐藏展开
#

默认代码块高度为 400，超出则隐藏。

  1. 准备一张向下展开的图片，放到 assets/icons 目录下
  2. 创建文件layouts\_partials\footer.html
 1<style>
 2    .highlight {
 3        /* 你可以根据需要调整这个高度 */
 4        max-height: 400px;
 5        overflow: hidden;
 6    }
 7
 8    .code-show {
 9        max-height: none !important;
10    }
11
12    .code-more-box {
13        width: 100%;
14        padding-top: 78px;
15        background-image: -webkit-gradient(linear, left top, left bottom, from(rgba(255, 255, 255, 0)), to(#fff));
16        position: absolute;
17        left: 0;
18        right: 0;
19        bottom: 0;
20        z-index: 1;
21    }
22
23    .code-more-btn {
24        display: block;
25        margin: auto;
26        width: 44px;
27        height: 22px;
28        background: #f0f0f5;
29        border-top-left-radius: 8px;
30        border-top-right-radius: 8px;
31        padding-top: 6px;
32        cursor: pointer;
33    }
34
35    .code-more-img {
36        cursor: pointer !important;
37        display: block;
38        margin: auto;
39        width: 22px;
40        height: 16px;
41    }
42
43</style>
44
45<script>
46    // 展开代码
47    function initCodeMoreBox() {
48        let codeBlocks = document.querySelectorAll(".highlight");
49        if (!codeBlocks) {
50            return;
51        }
52        codeBlocks.forEach(codeBlock => {
53            // 校验是否overflow
54            if (codeBlock.scrollHeight <= codeBlock.clientHeight) {
55                return;
56            }
57            // 展开元素初始化
58            // codeMoreBox
59            let codeMoreBox = document.createElement('div');
60            codeMoreBox.classList.add('code-more-box');
61            // codeMoreBtn
62            let codeMoreBtn = document.createElement('span');
63            codeMoreBtn.classList.add('code-more-btn');
64            // codeMoreImg
65            let codeMoreImg = document.createElement('img');
66            codeMoreImg.classList.add('code-more-img');
67            codeMoreImg.src = {{ (resources.Get "icons/codeMore.svg").Permalink }}
68            // 元素添加
69            codeMoreBtn.appendChild(codeMoreImg);
70            codeMoreBox.appendChild(codeMoreBtn);
71            codeBlock.appendChild(codeMoreBox)
72            // 监听事件
73            codeMoreBtn.addEventListener('click', () => {
74                // 展开代码块
75                codeBlock.classList.add('code-show');
76                // 隐藏更多按钮
77                codeMoreBox.style.display = 'none';
78                // 触发resize事件，重新计算目录位置
79                window.dispatchEvent(new Event('resize'))
80            })
81        })
82    }
83
84    initCodeMoreBox();
85
86</script>

顶部阅读进度条和百分比
#

 1<style>
 2    /* 顶部阅读进度条 */
 3    #top-scroll-bar {
 4        position: fixed;
 5        top: 0;
 6        left: 0;
 7        z-index: 9999;
 8        display: none;
 9        width: 0;
10        height: 3px;
11        background: #61b0ff;
12    }
13    /* 进度百分比显示 */
14    #progress-percentage{
15        position: fixed;
16        top: 10px;
17        right: 20px;
18        background: rgba(0, 0, 0, 0.7);
19        color: #fff;
20        padding: 5px 10px;
21        border-radius: 20px;
22        font-size: 14px;
23        z-index: 1001;
24        transition: opacity 0.3s;
25    }
26</style>
27
28<script>
29<!-- 进度条逻辑 -->
30    function initReadingProgressBar() {
31        window.addEventListener('scroll', () => {
32            // 获取DOM元素
33            const bar = document.getElementById('top-scroll-bar');
34            const progressPercentage = document.getElementById('progress-percentage');
35            if (!bar) return;
36            // 计算滚动进度
37            const scrollTop = document.documentElement.scrollTop || document.body.scrollTop;
38            const scrollHeight = document.documentElement.scrollHeight - document.documentElement.clientHeight;
39            const width = (scrollTop / scrollHeight) * 100;
40            bar.style.width = width + '%';
41            bar.style.display = 'block';
42
43            // 更新百分比显示
44            progressPercentage.textContent = Math.round(width) + "%";
45            // 根据滚动位置调整百分比显示的透明度
46            if (scrollTop > 100) {
47                progressPercentage.style.opacity = '1';
48            }else {
49                progressPercentage.style.opacity = '0.7';
50            }
51        });
52    }
53    
54   initReadingProgressBar();
55</script>
56<body>
57<!-- 进度条 -->
58<div id="top-scroll-bar"></div>
59<!-- 进度百分比显示 -->
60<div id="progress-percentage">0%</div>
61</body>

文章密码功能
#

将./themes/blowfish/layouts/_default/single.html复制到layouts/_default/single.html并编辑。

 1{{ define "main" }}
 2{{ .Scratch.Set "scope" "single" }}
 3
 4<article>
 5  {{ if .Params.showHero | default (.Site.Params.article.showHero | default false) }}
 6  {{ $heroStyle := .Params.heroStyle }}
 7  {{ if not $heroStyle }}{{ $heroStyle = .Site.Params.article.heroStyle }}{{ end }}
 8  {{ $heroStyle := print "hero/" $heroStyle ".html" }}
 9  {{ if templates.Exists ( printf "partials/%s" $heroStyle ) }}
10  {{ partial $heroStyle . }}
11  {{ else }}
12  {{ partial "hero/basic.html" . }}
13  {{ end }}
14  {{ end }}

在此处代码后面添加：

 1  <!-- 文章密码功能 -->
 2  {{ if ( .Params.password | default "" ) }}
 3  <script>
 4      (function(){
 5          if (prompt('请输入文章密码') != "{{ .Params.password }}"){
 6              alert('密码错误！');
 7              if (history.length === 1) {
 8                  window.opener = null;
 9                  window.open('', '_self');
10                  window.close();
11              } else {
12                  history.back();
13              }
14          }
15      })();
16  </script>
17  {{ end }}
18  <!-- 文章密码功能结束 -->
